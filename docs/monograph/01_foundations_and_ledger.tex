\subsection{The Four-Dimensional Yang--Mills Existence and Mass Gap Problem}
Let $G$ be a compact, connected, simple Lie group with Lie algebra $\mathfrak{g}$ (prototypically $G = SU(2)$ or $G = SU(3)$). In the Clay Mathematics Institute formulation \cite{JaffeWitten2000}, establishing four-dimensional quantum Yang--Mills theory requires constructing a nontrivial relativistic quantum field theory on Minkowski space $\R^{1,3}$ satisfying the Wightman axioms \cite{StreaterWightman1964}---or, equivalently via analytic continuation, a family of Euclidean Schwinger distributions $\{S_n\}_{n \ge 1} \subset \mathcal{S}'(\R^{4n})$ satisfying the Osterwalder--Schrader (OS) axioms \cite{OsterwalderSchrader1973,OsterwalderSchrader1975}:
\begin{itemize}
    \item \textbf{(OS0) Temperedness / Analyticity:} Each $S_n$ is a tempered distribution on $\R^{4n}$ (real-analytic away from coinciding points) satisfying linear growth bounds.
    \item \textbf{(OS1) Euclidean Covariance:} $S_n$ is invariant under the full Euclidean group $E(4) = \R^4 \rtimes O(4)$.
    \item \textbf{(OS2) Reflection Positivity:} Let $\theta(x_0, \mathbf{x}) = (-x_0, \mathbf{x})$ denote time reflection across the hyperplane $\Pi_0 = \{x_0 = 0\}$. For every finite sequence of test functions $f_n \in \mathcal{S}_+(\R^{4n})$ supported strictly in the positive-time open half-space $\{0 < x_1^0 < \dots < x_n^0\}$,
    \begin{equation}
    \sum_{m,n \ge 0} S_{m+n}(\Theta f_m \otimes f_n) \ge 0, \qquad (\Theta f_m)(x_1,\dots,x_m) := \overline{f_m(\theta x_m, \dots, \theta x_1)}.
    \end{equation}
    \item \textbf{(OS3) Symmetry:} $S_n(x_{\pi(1)}, \dots, x_{\pi(n)}) = S_n(x_1, \dots, x_n)$ for all permutations $\pi \in \mathfrak{S}_n$.
    \item \textbf{(OS4) Cluster Decomposition:} For any spatial or temporal translation $a \in \R^4$ with $|a| \to \infty$,
    \begin{equation}
    \lim_{|a| \to \infty} \left( S_{m+n}(f_m \otimes T_a f_n) - S_m(f_m) S_n(f_n) \right) = 0.
    \end{equation}
\end{itemize}
Furthermore, the physical Hamiltonian $H \ge 0$ reconstructed on the OS Hilbert space $\mathcal{H}$ must possess a unique vacuum vector $\Omega \in \mathcal{H}$ ($H\Omega = 0$) and a strictly positive spectral gap:
\begin{equation}
\spec(H) \subset \{0\} \cup [\Delta, \infty), \qquad \Delta > 0.
\end{equation}

\subsection{The Certified Five-Toll Corridor Ledger}
In constructive cluster expansions \cite{KoteckyPreiss1986,Brydges1986}, a strictly positive mass-gap floor $\Delta\kappa > 0$ is obtained by subtracting four entropic and geometric tolls from the bare polymer decay rate $\kappa_0 = -\ln z_0$:
\begin{equation}
\Delta\kappa := \kappa_0 - \underbrace{\big(-\ln(1 - \rho_{\mathrm{res}})\big)}_{\text{resolvent envelope}} - \underbrace{\ln 16}_{\text{4D fold toll}} - \underbrace{\ln(8e)}_{\text{lattice animal entropy}} - \underbrace{\rho_{\mathrm{res}}^2 \, \Sigma_{\mathrm{floor}}}_{\text{two-leg bridge overhead}},
\end{equation}
where $\rho_{\mathrm{res}} = \frac{1}{567}$ is the resolvent contraction ratio, $\Sigma_{\mathrm{floor}} = \frac{4319}{10} = 431.9$ is the summed two-leg bridge floor, and $L_{\mathrm{final}} = 8$ is the final coarse-block scale.

\begin{theorem}[Real-Valued Five-Toll Analytic Bounds, \texttt{AnalyticBounds.lean}]
\label{thm:five_toll_bounds}
The following strict inequalities hold over $\R$:
\begin{enumerate}
    \item \emph{Resolvent-Envelope Bound} (\texttt{envelope\_real\_bound}): $-\ln\!\left(1 - \frac{1}{567}\right) < \frac{177}{100000}$.
    \item \emph{Structural 4D Fold Toll} (\texttt{lnFold\_real\_bound}): $\ln 16 < \frac{27726}{10000}$.
    \item \emph{Lattice Animal Entropy Bound} (\texttt{lnC4\_real\_bound}): $\ln(8e) < \frac{6159}{2000}$.
    \item \emph{Two-Leg Bridge Overhead} (\texttt{bridge\_real\_bound}): $\left(\frac{1}{567}\right)^2 \cdot \frac{4319}{10} < \frac{84}{62500}$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) For every $x > 0$, $1 - x^{-1} \le \ln x$. Setting $x = \frac{566}{567}$ gives $-\ln(1 - \frac{1}{567}) \le \frac{1}{566} < \frac{177}{100000}$ since $100000 < 100182 = 177 \cdot 566$.
(2) Using $16 = 2^4$ and $\ln 2 < 0.6931471808$, $\ln 16 = 4\ln 2 < 2.7725887232 < \frac{27726}{10000}$.
(3) Using $8 = 2^3$, $\ln(8e) = 3\ln 2 + 1 < 3.0794415424 < \frac{6159}{2000}$.
(4) Direct rational arithmetic yields $4319 \cdot 62500 = 269937500 < 270050760 = 84 \cdot 567^2 \cdot 10$.
\end{proof}

\begin{corollary}[Real Five-Toll Surplus and Coarse-Block Mass Floor, \texttt{real\_five\_toll\_surplus}]
\label{cor:five_toll_surplus}
For any bare decay rate $\kappa_0 \ge \frac{61516}{10000} = 6.1516$, the certified five-toll decay exponent $\Delta\kappa$ strictly exceeds the corridor pair margin $m_{\mathrm{pair}} = \frac{29489277}{100000000}$, and the final coarse-block exponent $\mu_0(8) = 8\,\Delta\kappa$ satisfies $\mu_0(8) \ge 2$:
\begin{equation}
\Delta\kappa > \frac{61516}{10000} - \frac{177}{100000} - \frac{27726}{10000} - \frac{6159}{2000} - \frac{84}{62500} = 0.296386 > 0.29489277,
\end{equation}
and $8\,\Delta\kappa > 2.371088 \ge 2$.
\end{corollary}
